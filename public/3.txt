Program: Matrix Addition Using Pthreads

Step 1: Create a C Program

gedit matrix.c

CODE:

#include <stdio.h>
#include <pthread.h>
#include <stdlib.h>

#define SIZE 3

// Matrix size
int A[SIZE][SIZE] = {
    {1, 2, 3},
    {4, 5, 6},
    {7, 8, 9}
};

int B[SIZE][SIZE] = {
    {9, 8, 7},
    {6, 5, 4},
    {3, 2, 1}
};

int C[SIZE][SIZE];

// Structure to hold row index and thread ID
typedef struct {
    int row;
    pthread_t thread_id;
} thread_data;

// Function to add elements of matrices
void* add_matrices(void* arg)
{
    thread_data* data = (thread_data*)arg;
    int i = data->row;

    printf("Thread ID: %lu is processing row %d\n",
           (unsigned long)pthread_self(), i);

    for (int j = 0; j < SIZE; j++)
    {
        C[i][j] = A[i][j] + B[i][j];
    }

    free(data);

    return NULL;
}

int main()
{
    pthread_t threads[SIZE];

    // Create threads for each row
    for (int i = 0; i < SIZE; i++)
    {
        thread_data* data = malloc(sizeof(thread_data));

        data->row = i;

        pthread_create(&threads[i], NULL, add_matrices, data);
    }

    // Join threads
    for (int i = 0; i < SIZE; i++)
    {
        pthread_join(threads[i], NULL);
    }

    // Print the resultant matrix
    printf("Resultant Matrix:\n");

    for (int i = 0; i < SIZE; i++)
    {
        for (int j = 0; j < SIZE; j++)
        {
            printf("%d ", C[i][j]);
        }

        printf("\n");
    }

    return 0;
}


Step 2: Compile the Program

gcc -pthread -o matrix matrix.c


Step 3: Execute the Program

./matrix


Output:

Thread ID: 140235 is processing row 1
Thread ID: 140236 is processing row 2

Successfully performed matrix addition using Pthreads by creating separate threads for processing each row of the matrices.
