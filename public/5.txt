Program: Parallel Linear Search Using Pthreads

Step 1: Create a C Program

gedit parallel_search.c

CODE:

#include <stdio.h>
#include <pthread.h>

#define SIZE 100

// Array size
int data[SIZE];

// Index of the found element
int found_index = -1;

// Mutex for synchronizing access
pthread_mutex_t lock;

// Function to search for a number in a subarray
void* search(void* arg)
{
    int start = *(int*)arg;

    // Start index for this thread's subarray
    for (int i = start; i < start + SIZE / 4; i++)
    {
        if (data[i] == 50)
        {
            // Lock mutex before updating shared variable
            pthread_mutex_lock(&lock);

            found_index = i;

            // Unlock mutex after updating
            pthread_mutex_unlock(&lock);

            return NULL;
        }
    }

    return NULL;
}

int main()
{
    pthread_t threads[4];

    // Array for starting indices
    int thread_ids[4];

    // Initialize data array with values from 1 to 100
    for (int i = 0; i < SIZE; i++)
    {
        data[i] = i + 1;
    }

    // Initialize the mutex
    pthread_mutex_init(&lock, NULL);

    // Create 4 threads
    for (int i = 0; i < 4; i++)
    {
        thread_ids[i] = i * (SIZE / 4);

        pthread_create(&threads[i], NULL, search,
                       (void*)&thread_ids[i]);
    }

    // Wait for all threads to complete
    for (int i = 0; i < 4; i++)
    {
        pthread_join(threads[i], NULL);
    }

    // Print the result
    if (found_index != -1)
    {
        printf("Number 50 found at index: %d\n", found_index);
    }
    else
    {
        printf("Number 50 not found.\n");
    }

    // Destroy the mutex
    pthread_mutex_destroy(&lock);

    return 0;
}


Step 2: Compile the Program

gcc -pthread -o parallel_search parallel_search.c


Step 3: Execute the Program

./parallel_search


Output:

Number 50 found at index: 49


RESULT:

Successfully implemented and executed Parallel Linear Search using Pthreads by dividing an array of 100 elements into four parts and searching them simultaneously using four threads.
